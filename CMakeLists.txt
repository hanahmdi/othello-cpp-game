cmake_minimum_required(VERSION 3.12)
project(OthelloCpp VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(othello_engine src/Othello.cpp)
target_include_directories(othello_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(othello src/main.cpp)
target_link_libraries(othello PRIVATE othello_engine)
include(CTest)
if(BUILD_TESTING)
    add_executable(othello_tests tests/test_othello.cpp)
    target_link_libraries(othello_tests PRIVATE othello_engine)
    add_test(NAME engine_tests COMMAND othello_tests)
endif()
